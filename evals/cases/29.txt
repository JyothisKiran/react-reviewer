import { createSlice, type PayloadAction } from "@reduxjs/toolkit";
import type { RootState } from "@/app/store";

type FiltersState = { department: string | null; showArchived: boolean };

const initialState: FiltersState = { department: null, showArchived: false };

const filtersSlice = createSlice({
  name: "filters",
  initialState,
  reducers: {
    departmentChanged(state, action: PayloadAction<string | null>) {
      state.department = action.payload;
    },
    archivedToggled(state) {
      state.showArchived = !state.showArchived;
    },
    filtersReset: () => initialState,
  },
});

export const { departmentChanged, archivedToggled, filtersReset } = filtersSlice.actions;
export const selectFilters = (state: RootState) => state.filters;
export default filtersSlice.reducer;
